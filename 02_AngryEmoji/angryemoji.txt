import cairo
import math

# 1. Setup Canvas/Surface
width, height = 500, 500
surface = cairo.ImageSurface(cairo.FORMAT_ARGB32, width, height)
ctx = cairo.Context(surface)

# Background putih
ctx.set_source_rgb(1, 1, 1)
ctx.paint()

# 5. Badan (Line to, garis hitam, diisi warna KUNING sama seperti wajah)
ctx.move_to(190, 330)
ctx.line_to(120, 500)
ctx.line_to(380, 500)
ctx.line_to(310, 330)
ctx.close_path()
ctx.set_source_rgb(1, 0.9, 0) # Warna kuning (Sama seperti wajah)
ctx.fill_preserve()
ctx.set_source_rgb(0, 0, 0)
ctx.set_line_width(6)
ctx.set_line_join(cairo.LINE_JOIN_ROUND)
ctx.stroke()

# 6. Tangan (Terlipat masuk ke dalam badan)
ctx.set_line_width(6)
ctx.set_line_cap(cairo.LINE_CAP_ROUND)

# Tangan kiri
ctx.move_to(195, 360)
ctx.line_to(240, 390)
ctx.curve_to(248, 400, 248, 410, 240, 420)
ctx.line_to(205, 435)
ctx.stroke()

# Tangan kanan
ctx.move_to(305, 360)
ctx.line_to(260, 390)
ctx.curve_to(252, 400, 252, 410, 260, 420)
ctx.line_to(295, 435)
ctx.stroke()

# 3. Wajah (Lingkaran kuning dengan Arc)
ctx.arc(250, 220, 140, 0, 2*math.pi)
ctx.set_source_rgb(1, 0.9, 0) # Warna kuning
ctx.fill_preserve()
ctx.set_source_rgb(0, 0, 0)
ctx.set_line_width(6)
ctx.stroke()

# 2. Mata besar tanpa pupil (Sisi kiri layar / X=170)
ctx.save()
ctx.translate(170, 200)
ctx.scale(1, 1.6)
ctx.arc(0, 0, 35, 0, 2*math.pi)
ctx.restore()
ctx.set_source_rgb(1, 1, 1)
ctx.fill_preserve()
ctx.set_source_rgb(0, 0, 0)
ctx.set_line_width(6)
ctx.stroke()

# 1. Mata kecil tanpa pupil (Sisi kanan layar / X=330)
ctx.arc(330, 200, 25, 0, 2*math.pi)
ctx.set_source_rgb(1, 1, 1)
ctx.fill_preserve()
ctx.set_source_rgb(0, 0, 0)
ctx.set_line_width(6)
ctx.stroke()

# 8. Alis marah
ctx.set_line_width(8)
# Alis kiri
ctx.move_to(130, 120)
ctx.line_to(220, 175)
ctx.stroke()
# Alis kanan
ctx.move_to(370, 140)
ctx.line_to(290, 175)
ctx.stroke()

# 4. Mulut segitiga sama sisi (Diperkecil ukurannya, puncak lancip di atas)
ctx.move_to(250, 275) # Puncak lancip atas
ctx.line_to(270, 310) # Kanan bawah
ctx.line_to(230, 310) # Kiri bawah
ctx.close_path()
ctx.set_source_rgb(0.9, 0.1, 0.1)
ctx.fill_preserve()
ctx.set_source_rgb(0, 0, 0)
ctx.set_line_width(6)
ctx.set_line_join(cairo.LINE_JOIN_ROUND)
ctx.stroke()

# 7. Icon Marah (8 rectangle, lancip saling bertemu ke dalam)
ctx.set_source_rgb(0.9, 0.1, 0.1)
cx, cy = 140, 90
w, t, g = 18, 6, 2

ctx.rectangle(cx - g - w, cy - g - t, w, t)
ctx.rectangle(cx - g - t, cy - g - w, t, w)
ctx.rectangle(cx + g, cy - g - t, w, t)
ctx.rectangle(cx + g, cy - g - w, t, w)
ctx.rectangle(cx - g - w, cy + g, w, t)
ctx.rectangle(cx - g - t, cy + g, t, w)
ctx.rectangle(cx + g, cy + g, w, t)
ctx.rectangle(cx + g, cy + g, t, w)

ctx.fill_preserve()

ctx.set_source_rgb(0, 0, 0)
ctx.set_line_width(2)
ctx.stroke()

# Export hasil
surface.write_to_png("emoji_marah_pycairo.png")
print("Berhasil memperkecil ukuran mulut emoji!")
